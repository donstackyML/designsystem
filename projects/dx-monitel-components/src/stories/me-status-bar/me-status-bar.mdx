## API

### StatusType
```typescript
type StatusType = 'error' | 'warning' | 'success' | 'info';
```

### StatusBarItem
```typescript
import {StatusType} from "./me-status-bar.component";

interface StatusBarItem {
  text: string;             // Текст
  textColor?: string;       // Цвет текста
  type?: StatusType;        // Тип статуса
  icon?: string;            // Иконка
  iconColor?: string;       // Цвет иконки
  showStatusIcon?: boolean; // Показать иконку, которая соответствует типу статуса
  fill?: boolean;           // Использовать заливку фона
  backgroundColor?: string; // Цвет фона
  readonly?: boolean;       // Только для чтения
  onClick?: () => void;     // Обработчик события клика
}
```
