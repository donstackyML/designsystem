# MeShiftSettings

Компонент `me-shift-settings` используется для настройки сдвига времени в компоненте `me-time-range`. Позволяет выбирать между абсолютным и относительным режимами, а также настраивать параметры сдвига времени.

### Смешанный режим

<Story of={shiftSettingsStories.Default} />

<Source
  code={`
// Template
<me-shift-settings
  [title]="'Начало периода'"
  [shiftType]="'mixed'"
  [currentShiftIsActive]="true"
  [absoluteDate]="startDate"
  [properties]="startShiftProperties"
  [propertiesVariant]="'full'"
  [switchTitle]="'Относительно текущего времени'"
  [switchEnabled]="true"
  (settingsChange)="onStartShiftSettingsChange($event)"
></me-shift-settings>

// Component
const startShiftProperties = [

  { text: 'Часы', key: 'hours', enabled: true, value: -1 },
  { text: 'Минуты', key: 'minutes', enabled: false, value: 0 },
  { text: 'Секунды', key: 'seconds', enabled: false, value: 0 },
];

startDate = new Date();

onStartShiftSettingsChange(settings: ShiftSettingsOutput) {
  // Обработка изменений настроек
}
  `}

### ShiftSettingsOutput

```typescript
export interface ShiftSettingsOutput {
  properties: Array<TimeShiftProperty> | MinimalTimeShiftProperty | null;
  absoluteDate: Date | string | number | null;
  isRelativeModeActive: boolean;
  shiftType: ShiftType;
}
```

### ShiftType

```typescript
export type ShiftType = 'relative' | 'absolute' | 'mixed';
```

### TimeShiftProperty

```typescript
export interface TimeShiftProperty {
  text: string;
  key: string;
  enabled: boolean;
  value: number;
}
```

### MinimalTimeShiftProperty

```typescript
export interface MinimalTimeShiftProperty {
  value: number;
  selectedUnit: string;
  displayExpr?: string;
  valueExpr?: string;
  units?: Array<{
    text?: string;
    value: string;
  }>;
  [key: string]: any;
}
```
