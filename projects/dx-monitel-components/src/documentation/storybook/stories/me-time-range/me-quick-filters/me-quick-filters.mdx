### С пользовательскими фильтрами

<Story of={quickFiltersStories.WithCustomFilters} />

<Source
  code={`
// Template
<me-quick-filters
  [title]="'Пользовательские фильтры'"
  [filters]="customFilters"
  [selectedFilterId]="selectedFilterId"
  (filterSelected)="onFilterSelected($event)"
></me-quick-filters>

// Component
const customFilters = [
  { id: 'today', text: 'Сегодня' },
  { id: 'yesterday', text: 'Вчера' },
  { id: 'this-week', text: 'Эта неделя' },
  { id: 'last-week', text: 'Прошлая неделя' },
  { id: 'this-month', text: 'Этот месяц' },
];

selectedFilterId = 'today';

onFilterSelected(filterId: string) {
  this.selectedFilterId = filterId;
}
  `}
  language="typescript"
  dark
/>
