# MeShiftProperties

Компонент `me-shift-properties` используется для настройки свойств временного сдвига в компоненте `me-time-range`. Позволяет настраивать сдвиг по различным единицам времени (месяцы, недели, дни, часы, минуты, секунды).

### С пользовательскими свойствами

<Story of={shiftPropertiesStories.WithCustomProperties} />

<Source
  code={`
// Template
<me-shift-properties
  [title]="'Пользовательские настройки'"
  [properties]="customProperties"
  (shiftPropertiesChange)="onShiftPropertiesChange($event)"
></me-shift-properties>

// Component
const customProperties = [
  { text: 'Секунды', key: 'seconds', enabled: true, value: 30 },
  { text: 'Минуты', key: 'minutes', enabled: true, value: 15 },
  { text: 'Часы', key: 'hours', enabled: false, value: 0 },
  { text: 'Дни', key: 'days', enabled: false, value: 0 },
];

onShiftPropertiesChange(properties: TimeShiftProperty[]) {
  // Обработка изменений свойств
}
  `}
  language="typescript"
  dark
/>
